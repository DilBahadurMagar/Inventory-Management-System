-- ==========================================
-- INVENTORY & ASSET MANAGEMENT SYSTEM SCHEMA
-- Database: PostgreSQL (3NF)
-- ==========================================

-- PHASE 1: INDEPENDENT LOOKUP TABLES (NO DEPENDENCIES)
-- These must be created first.

CREATE TABLE roles (
    role_id SERIAL PRIMARY KEY,
    role_name VARCHAR(50) UNIQUE NOT NULL,
    description TEXT
);

CREATE TABLE categories (
    category_id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    description TEXT
);

CREATE TABLE locations (
    location_id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    address TEXT,
    is_active BOOLEAN DEFAULT TRUE
);

CREATE TABLE suppliers (
    supplier_id SERIAL PRIMARY KEY,
    company_name VARCHAR(150) NOT NULL,
    contact_name VARCHAR(100),
    email VARCHAR(255),
    phone VARCHAR(20)
);

-- PHASE 2: CORE ENTITY TABLES 
-- These depend on the lookup tables above.

CREATE TABLE users (
    user_id SERIAL PRIMARY KEY,
    role_id INT REFERENCES roles(role_id) ON DELETE SET NULL,
    username VARCHAR(50) UNIQUE NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    password_hash TEXT NOT NULL, -- Store hashed passwords only
    full_name VARCHAR(100),
    is_active BOOLEAN DEFAULT TRUE,
    last_login TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE items (
    item_id SERIAL PRIMARY KEY,
    category_id INT REFERENCES categories(category_id) ON DELETE SET NULL,
    name VARCHAR(200) NOT NULL,
    sku VARCHAR(50) UNIQUE NOT NULL,
    reorder_level INT DEFAULT 10,
    unit_price DECIMAL(10, 2) DEFAULT 0.00
);

-- PHASE 3: TRACKING TABLES
-- These manage physical stock and individual serialized assets.

CREATE TABLE inventory (
    inventory_id SERIAL PRIMARY KEY,
    item_id INT REFERENCES items(item_id) ON DELETE CASCADE,
    location_id INT REFERENCES locations(location_id) ON DELETE CASCADE,
    quantity_on_hand INT NOT NULL DEFAULT 0,
    last_updated TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE assets (
    asset_id SERIAL PRIMARY KEY,
    item_id INT REFERENCES items(item_id) ON DELETE CASCADE,
    serial_number VARCHAR(100) UNIQUE,
    location_id INT REFERENCES locations(location_id) ON DELETE SET NULL,
    supplier_id INT REFERENCES suppliers(supplier_id) ON DELETE SET NULL,
    status VARCHAR(50) DEFAULT 'In Stock', -- e.g., 'In Use', 'Maintenance', 'Retired'
    purchase_date DATE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- PHASE 4: AUDIT & LOGGING
-- Captures every movement for history and accountability.

CREATE TABLE transactions (
    transaction_id SERIAL PRIMARY KEY,
    item_id INT REFERENCES items(item_id) ON DELETE SET NULL,
    location_id INT REFERENCES locations(location_id) ON DELETE SET NULL,
    user_id INT REFERENCES users(user_id) ON DELETE SET NULL,
    transaction_type VARCHAR(20) NOT NULL, -- e.g., 'STOCK_IN', 'STOCK_OUT', 'TRANSFER'
    quantity INT NOT NULL,
    notes TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
